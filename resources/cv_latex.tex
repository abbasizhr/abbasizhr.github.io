\section*{Selected Research Projects}

\entry{Fresnel Diffraction from Phase-Gradient Singularities}{Journal research project}
{Institute for Advanced Studies in Basic Sciences (IASBS)}
{
\begin{itemize}
  \item Investigated phase-gradient singularity as a physical origin of Fresnel diffraction in optical phase objects.
  \item Modeled diffracted intensity distributions for Fresnel double mirror and Fresnel biprism systems using Fresnel--Kirchhoff diffraction theory.
  \item Compared theoretical results with experimental diffraction patterns and demonstrated applications in optical metrology and simultaneous wavelength/angle measurement.
\end{itemize}
}

\entry{CCD-Based Sub-Pixel Resolution Enhancement and Computational Imaging}{Journal research project}
{University of Zanjan}
{
\begin{itemize}
  \item Developed a CCD-based sub-pixel resolution approach for reconstructing higher-resolution images from multiple low-resolution frames captured under fine relative motion.
  \item Applied MATLAB-based image reconstruction and quantitative error analysis to evaluate the enhanced image against a high-resolution reference.
  \item Connected optical resolution enhancement with computational imaging, image processing, and super-resolution concepts.
\end{itemize}
}

\entry{Fresnel Double-Mirror Optical Metrology}{Conference research project}
{Institute for Advanced Studies in Basic Sciences (IASBS)}
{
\begin{itemize}
  \item Analyzed a Fresnel diffraction-based method for simultaneous measurement of incident light wavelength and Fresnel double-mirror apex angle.
  \item Used Fresnel--Kirchhoff diffraction modeling, fringe-spacing analysis, and camera-based intensity measurements.
  \item Validated the method experimentally using a He--Ne laser setup, estimating the laser wavelength at approximately 633 nm.
\end{itemize}
}

\entry{AI-Assisted Diffraction and Fringe Pattern Analysis}{Ongoing research direction}
{Independent computational imaging project}
{
\begin{itemize}
  \item Extending prior work on Fresnel diffraction, CCD imaging, and optical pattern analysis toward AI-assisted computational imaging.
  \item Developing workflows for fringe-pattern interpretation, feature extraction, denoising, and parameter estimation from optical images.
\end{itemize}
}
